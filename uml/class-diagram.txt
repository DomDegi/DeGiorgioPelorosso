' ==========================================
' THE CENTRAL HUB
' ==========================================
class Orchestrator <<Module>> {
    + orchestrator(batch_size, input, output, rules, sensors): void
}

' ==========================================
' ABSTRACT INTERFACES (The Inner Ring)
' ==========================================
interface ITelemetryReader {
    + extract_batch(batch_size: int): pl.DataFrame
}

interface IRulesEngine {
    + evaluate_rules(batch: pl.DataFrame): Tuple
}

interface IStateMemory {
    + get_current_count(rule_id, sensor_id): int
    + set_consecutive_count(...)
    + get_last_value(...)
    + set_last_value(...)
}

interface IOutputWriter {
    + write_valid_batch(valid_data: pl.DataFrame): void
    + write_alarms_batch(alarm_data: pl.DataFrame): void
}

' ==========================================
' CONCRETE IMPLEMENTATIONS (The Outer Ring)
' ==========================================
class CSVTelemetryReader {
    - _buffer: pl.DataFrame
    - _batched_reader: BatchedCsvReader
    + extract_batch(batch_size: int): pl.DataFrame
}

class PolarsRulesEngine {
    - rules: list
    + evaluate_rules(batch: pl.DataFrame): Tuple
}

class DictStateMemory {
    - _consecutive_counts: dict
    - _last_values: dict
    + get_current_count(rule_id, sensor_id): int
    + set_consecutive_count(...)
    + get_last_value(...)
    + set_last_value(...)
}

class CSVOutputWriter {
    - valid_file_path: str
    - alarms_file_path: str
    + write_valid_batch(valid_data: pl.DataFrame): void
    + write_alarms_batch(alarm_data: pl.DataFrame): void
}

' ==========================================
' RADIAL RELATIONSHIPS & ROUTING (The Star)
' ==========================================

' 1. UP SPOKE: Reader
Orchestrator -up-> ITelemetryReader : " Ingests via"
CSVTelemetryReader .right.|> ITelemetryReader

' 2. DOWN SPOKE: Rules Engine
Orchestrator -down-> IRulesEngine : " Evaluates via"
PolarsRulesEngine .up.|> IRulesEngine

' 3. RIGHT SPOKE: State Memory
Orchestrator -right-> IStateMemory : " Initializes"
DictStateMemory .down.|> IStateMemory

' 4. LEFT SPOKE: Writer
Orchestrator -left-> IOutputWriter : " Exports via"
CSVOutputWriter .up.|> IOutputWriter

' 5. CROSS-DEPENDENCY (Outer Ring to Inner Ring)
' Using "o-up->" fights the gravity of Graphviz, forcing IStateMemory to stay
' on the middle horizontal plane instead of being dragged to the bottom.
PolarsRulesEngine o-up-> IStateMemory : " Tracks state via"
